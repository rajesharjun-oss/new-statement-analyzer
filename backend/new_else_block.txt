        else:
            if not current:
                continue

            # --- NEW: merge split decimal tails into debit/credit/balance BEFORE appending narration ---
            # Handles:
            #   "13,046,880." + "13"  -> "13,046,880.13"
            #   "14,156,559.6" + "7"  -> "14,156,559.67"
            def _collect_digit_tokens(rr):
                tokens = []
                # Sometimes the orphan digits land in Remarks/Reference due to column jitter
                for k in ("Debit", "Credit", "Balance", "Reference", "Remarks", "ValueDate"):
                    v = (rr.get(k) or "").strip()
                    if not v:
                        continue
                    for t in v.split():
                        if re.fullmatch(r"\d{1,2}", t):
                            tokens.append(t)
                return tokens

            tails = _collect_digit_tokens(r)

            def _try_merge_tail(base, tails):
                base = (base or "").strip()
                if not tails:
                    return base, tails

                # Case A: ends with "." → needs 2 decimal digits
                if re.search(r"\.\s*$", base):
                    if len(tails[0]) == 2:
                        return base + tails[0], tails[1:]
                    # two 1-digit tails e.g. "1" "3" => "13"
                    if len(tails) >= 2 and len(tails[0]) == 1 and len(tails[1]) == 1:
                        return base + tails[0] + tails[1], tails[2:]
                    return base, tails

                # Case B: ends with ".<digit>" → needs 1 more digit
                if re.search(r"\.\d\s*$", base):
                    return base + tails[0], tails[1:]

                return base, tails

            # Merge into the monetary fields if needed
            for field in ("debit", "credit", "balance"):
                merged, tails = _try_merge_tail(current.get(field, ""), tails)
                current[field] = merged

            # --- EXISTING: continuation line → append narration/reference ---
            ref = (r.get("Reference") or "").strip()
            rem = (r.get("Remarks") or "").strip()
            if ref:
                current["reference"] = (current["reference"] + " " + ref).strip()
            if rem:
                current["remarks"] = (current["remarks"] + " " + rem).strip()
